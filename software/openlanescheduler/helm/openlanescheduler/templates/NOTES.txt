Useful Commands:

  # View all pods
  kubectl get pods -n {{ .Release.Namespace }} -l "app.kubernetes.io/instance={{ .Release.Name }}"

  # View backend logs
  kubectl logs -n {{ .Release.Namespace }} -l "app.kubernetes.io/component=backend" -f

  # View Redis status
  kubectl get statefulset -n {{ .Release.Namespace }} {{ include "openlanescheduler.redis.fullname" . }}

  # Scale backend
  kubectl scale deployment -n {{ .Release.Namespace }} {{ include "openlanescheduler.backend.fullname" . }} --replicas=3

  # Restart backend
  kubectl rollout restart deployment -n {{ .Release.Namespace }} {{ include "openlanescheduler.backend.fullname" . }}
